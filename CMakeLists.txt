cmake_minimum_required(VERSION 3.21)
project(LabelStudioDesktop VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_compile_options(/utf-8)
endif()
find_package(Qt6 6.6.3 REQUIRED COMPONENTS Core Gui Network Quick QuickControls2 Xml Test LinguistTools)
qt_standard_project_setup(REQUIRES 6.6)
add_library(annotation_core STATIC src/annotationdocument.h src/annotationdocument.cpp)
target_link_libraries(annotation_core PUBLIC Qt6::Core Qt6::Gui)
target_include_directories(annotation_core PUBLIC src)
add_library(desktop_api STATIC src/apiclient.h src/apiclient.cpp src/projectapi.cpp src/taskworkflow.cpp src/projectmanagement.cpp src/projectcatalog.h src/projectcatalog.cpp src/accountmanager.h src/accountmanager.cpp src/datamanager.h src/datamanager.cpp)
target_link_libraries(desktop_api PUBLIC annotation_core Qt6::Network)
add_library(desktop_config STATIC src/labelingconfig.h src/labelingconfig.cpp)
target_link_libraries(desktop_config PUBLIC desktop_api Qt6::Quick Qt6::Xml)
add_library(desktop_i18n STATIC src/languagemanager.h src/languagemanager.cpp)
target_include_directories(desktop_i18n PUBLIC src)
target_link_libraries(desktop_i18n PUBLIC Qt6::Qml)
qt_add_translation(DESKTOP_QM_FILES translations/desktop_zh_CN.ts)
get_filename_component(QT_TRANSLATION_DIR "${Qt6_DIR}/../../../translations" ABSOLUTE)
foreach(CATALOG qtbase_zh_CN qtdeclarative_zh_CN)
    configure_file("${QT_TRANSLATION_DIR}/${CATALOG}.qm" "${CMAKE_CURRENT_BINARY_DIR}/${CATALOG}.qm" COPYONLY)
endforeach()
qt_add_resources(desktop_i18n desktop_translations PREFIX "/i18n" BASE "${CMAKE_CURRENT_BINARY_DIR}"
    FILES ${DESKTOP_QM_FILES} "${CMAKE_CURRENT_BINARY_DIR}/qtbase_zh_CN.qm" "${CMAKE_CURRENT_BINARY_DIR}/qtdeclarative_zh_CN.qm")
include(CTest)
qt_add_resources(QML_RESOURCES resources.qrc assets/template-catalog.qrc)
qt_add_executable(LabelStudioDesktop WIN32 src/main.cpp ${QML_RESOURCES})
target_link_libraries(LabelStudioDesktop PRIVATE desktop_config desktop_i18n Qt6::Quick Qt6::QuickControls2)
install(TARGETS LabelStudioDesktop RUNTIME DESTINATION .)
if(BUILD_TESTING)
    qt_add_executable(tst_language tests/tst_language.cpp)
    set_target_properties(tst_language PROPERTIES WIN32_EXECUTABLE FALSE)
    target_link_libraries(tst_language PRIVATE desktop_i18n desktop_api Qt6::Test)
    add_test(NAME language COMMAND tst_language)
    qt_add_executable(tst_document tests/tst_document.cpp)
    set_target_properties(tst_document PROPERTIES WIN32_EXECUTABLE FALSE)
    target_link_libraries(tst_document PRIVATE annotation_core Qt6::Test)
    add_test(NAME document COMMAND tst_document)
    qt_add_executable(tst_api tests/tst_api.cpp)
    set_target_properties(tst_api PROPERTIES WIN32_EXECUTABLE FALSE)
    target_link_libraries(tst_api PRIVATE desktop_api Qt6::Test)
    add_test(NAME api COMMAND tst_api)
    qt_add_executable(tst_ui tests/tst_ui.cpp ${QML_RESOURCES})
    set_target_properties(tst_ui PROPERTIES WIN32_EXECUTABLE FALSE)
    target_link_libraries(tst_ui PRIVATE desktop_config desktop_i18n Qt6::Quick Qt6::QuickControls2 Qt6::Test)
    add_test(NAME native_ui COMMAND tst_ui)
    qt_add_executable(tst_live tests/tst_live.cpp ${QML_RESOURCES})
    set_target_properties(tst_live PROPERTIES WIN32_EXECUTABLE FALSE)
    target_link_libraries(tst_live PRIVATE desktop_config desktop_i18n Qt6::Quick Qt6::QuickControls2 Qt6::Test)
    add_test(NAME live_django COMMAND tst_live)
    set_tests_properties(native_ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
    if(WIN32 AND EXISTS "$ENV{WINDIR}/Fonts/msyh.ttc")
        file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/test-fonts")
        file(COPY "$ENV{WINDIR}/Fonts/msyh.ttc" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/test-fonts")
        set_property(TEST native_ui APPEND PROPERTY ENVIRONMENT "QT_QPA_FONTDIR=${CMAKE_CURRENT_BINARY_DIR}/test-fonts")
    endif()
endif()
